 57    def setUp(self):
 58        """Antes de cada test crea una base de datos falsa en una carpeta
 59        temporal del ordenador con la tabla usuarios vacía"""
 60        self.tmp_dir = tempfile.mkdtemp()
 61        self.db_path = Path(self.tmp_dir) / "biblioteca.db"
 62
 63        con = sqlite3.connect(self.db_path)
 64        con.execute(
 65            """
 66            CREATE TABLE usuarios (
 67                id        INTEGER PRIMARY KEY AUTOINCREMENT,
 68                nombre    TEXT    NOT NULL,
 69                apellidos TEXT    NOT NULL,
 70                email     TEXT    NOT NULL,
 71                habilitado INTEGER NOT NULL DEFAULT 1
 72            )
 73            """
 74        )
 75        con.commit()
 76        con.close()
 77
 78        self.patcher = patch("DAO.UsuarioDAO.RUTA_BD", self.db_path)
 79        self.patcher.start()
 80
 81        self.dao = UsuarioDAO()
 82
 83    def tearDown(self):
 84        """Después de cada test borra la carpeta temporal. Si Windows no la
 85        deja borrar porque tiene el archivo bloqueado, espera un momento y
 86        lo intenta de nuevo. Si sigue sin poder, lo deja pasar para que no
 87        falle el test por culpa del sistema operativo"""
 88        self.patcher.stop()
 89
 90        if hasattr(self, 'dao'):
 91            del self.dao
 92
 93        try:
 94            shutil.rmtree(self.tmp_dir)
 95        except PermissionError:
